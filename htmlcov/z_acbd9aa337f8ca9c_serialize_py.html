19def dump_spec(spec: FlowSpec) -> dict[str, Any]: 

20 """Dump a flow spec to its canonical wire representation. 

21 

22 The returned dict is JSON-compatible and stable: re-dumping a reloaded spec 

23 reproduces it exactly, i.e. 

24 `dump_spec(load_spec_data(dump_spec(spec))) == dump_spec(spec)`. Fields that 

25 are unset — or explicitly set to their default value — are omitted, so 

26 effective values always survive a round trip but `model_fields_set` 

27 membership does not. Explicit `None` is preserved, and Flow's `NotGiven` 

28 sentinel never appears. Live values normalize to references (a 

29 registered callable to its registry name, a registry object such as a tool 

30 in agent args to its reference dict), so a reloaded spec is wire-equivalent 

31 to, but not necessarily object-equal to, the original. The runner resolves 

32 a name back to a callable only in `factory` fields and `store.filter`; in 

33 any other position (e.g. factory `args`) the name reloads as a plain 

34 string — `validate_portable_spec` flags these before dumping. Typed values 

35 are normalized through validation, so a coercible value assigned without 

36 validation (e.g. a numeric string in a float field) dumps in canonical form. 

37 

38 Values in untyped fields (e.g. `flow_metadata`, `model_args`) should be 

39 JSON-native; anything else follows pydantic's JSON coercion (a tuple becomes 

40 a list, a `datetime` a string, a set a list in unspecified order) and may 

41 not round-trip. In fields that skip 

42 validation (e.g. `early_stopping`), serializable live objects carry 

43 one-way: they reload as plain data. Arguments captured in a registered 

44 object's params are encoded by inspect-ai at construction time 

45 (non-serializable values are already stringified before Flow sees them). 

46 Whether a reference resolves in the target process — installed packages, 

47 shipped files — is a separate concern; gate portable use with 

48 `validate_portable_spec` before dumping. YAML/JSON rendering is left to the 

49 caller. 

50 

51 Args: 

52 spec: The flow spec to dump. 

53 

54 Returns: 

55 A JSON-compatible dict representation of the spec. 

56 

57 Raises: 

58 ValueError: If the spec cannot be reconstructed from its dump: a live 

59 Inspect object in a typed field, an unregistered callable that 

60 pydantic cannot serialize as data such as a plain function or 

61 lambda (a callable *object* with serializable fields in an untyped 

62 position instead follows the data-coercion rule above and reloads 

63 as plain data), a non-finite float, or a `NotGiven` sentinel in an 

64 untyped mapping (including a user value of that exact shape, which 

65 is indistinguishable on the wire). 

66 """ 

67 # JSON-mode serialization silently degrades non-finite floats (kept in 

68 # FlowBase fields, nulled in untyped mappings, dropped entirely by nested 

69 # inspect-ai models), so check the original python values first. This pass 

70 # sees raw user objects, so it needs the strict fallback too — the lenient 

71 # one assumes every callable has __code__ (builtins and partials do not) 

72 _reject_invalid_wire_data( 

73 model_dump(spec, mode="python", fallback=_wire_fallback), path="" 

74 ) 

75 data = model_dump(spec, fallback=_wire_fallback) 

76 # return the re-dump of the re-validated spec, so the output is canonical 

77 # by construction: a coercible value assigned without validation (pydantic 

78 # does not validate on assignment) normalizes instead of leaking its 

79 # non-canonical form onto the wire 

80 verified = _verify_reloadable(data) 

81 # re-validation itself coerces (e.g. an unvalidated "1e999" becomes inf), 

82 # so check the verified values as well — symmetric with load_spec_data 

83 _reject_invalid_wire_data(model_dump(verified, mode="python"), path="") 

84 canonical = model_dump(verified, fallback=_wire_fallback) 

85 # invariant on the exact returned value (the earlier walks inspect 

86 # intermediate forms, whose serialization pydantic may change) 

87 _reject_invalid_wire_data(canonical, path="") 

88 return canonical

91def load_spec_data(data: Mapping[str, Any]) -> FlowSpec: 

92 """Reconstruct a flow spec from its canonical wire representation. 

93 

94 Inverse of `dump_spec()`. Validation is strict: unknown fields are rejected 

95 (except in `FlowFactory` mappings, whose API forwards unrecognized keys as 

96 factory arguments), as is an embedded `NotGiven` sentinel or a non-finite 

97 float (whether present literally or produced by coercing a string such as 

98 `"NaN"`). JSON and YAML parsers accept both, but they signal data that did 

99 not come from `dump_spec()`, and a sentinel would be misread as a user 

100 value. 

101 

102 Args: 

103 data: Spec data produced by `dump_spec()`, possibly after a JSON or 

104 YAML round trip. 

105 

106 Returns: 

107 The reconstructed flow spec. 

108 

109 Raises: 

110 ValueError: If the data contains a `NotGiven` sentinel or a non-finite 

111 float, or fails spec validation. 

112 """ 

113 _reject_invalid_wire_data(data, path="") 

114 spec = FlowSpec.model_validate(data, extra="forbid") 

115 # pydantic coerces numeric strings, so a value like "1e999" only becomes a 

116 # non-finite float after validation; check the validated values as well 

117 _reject_invalid_wire_data(model_dump(spec, mode="python"), path="") 

118 return spec

121def _verify_reloadable(data: dict[str, Any]) -> FlowSpec: 

122 # A live object created by a registry factory (e.g. an @task/@solver 

123 # instance) serializes to a registry dict that is fine in an untyped `args` 

124 # mapping but does not fit a typed field like `tasks` or `solver`. Position 

125 # is invisible to _wire_fallback, so enforce the round-trip contract by 

126 # confirming the dump actually re-validates. This also covers any future 

127 # typed field that accepts live objects. 

128 try: 

129 return FlowSpec.model_validate(data, extra="forbid") 

130 except ValidationError as e: 

131 raise ValueError( 

132 "The spec cannot be reconstructed in another process: its dump does " 

133 "not re-validate. This usually means a typed field holds a live " 

134 "Inspect object (e.g. an instantiated Task, Model, Solver, Scorer, " 

135 "or Agent) whose serialized form does not fit that field; replace it " 

136 "with a Flow type or a registry/file name reference. Affected " 

137 f"field(s): {', '.join(_affected_fields(e))}." 

138 ) from e
