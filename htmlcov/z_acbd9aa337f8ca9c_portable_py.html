108def validate_portable_spec(spec: FlowSpec) -> None: 

109 """Validate that a flow spec can be serialized and recreated in another process. 

110 

111 A portable spec survives the boundary the venv runner and remote 

112 orchestrators cross: dumped to YAML/JSON, then re-validated as a 

113 `FlowSpec` in a fresh process. Values that do not survive it are live 

114 (already-instantiated) Inspect objects such as `Task`, `Model`, `Scorer`, 

115 `Solver`, and `Agent` — which reload as a `repr` string or a plain dict 

116 rather than the object — and callables the child cannot resolve. A 

117 serialized callable is looked up in the Inspect registry, so only a 

118 registered object (or a registry name string) is portable: an undecorated 

119 function is not, even at module level, and neither are lambdas, 

120 `functools.partial`, nested functions, classes, and callable objects. Any 

121 JSON-serializable data is portable. 

122 

123 Every field is checked, including free-form containers (`args`, 

124 `metadata`, `flow_metadata`, scanner params) and any inline `FlowSpec` in 

125 `includes` — a path string there is not loaded, so an included file's 

126 contents are not checked. The spec is not expanded or resolved, and nothing 

127 is installed or launched. 

128 

129 Four limitations. A value Pydantic coerces natively on dump is reported 

130 as portable, because it reloads as the coerced type rather than being 

131 lost: `tuple` and `set` become `list`, a dataclass or `BaseModel` becomes 

132 `dict`, non-string mapping keys become strings, `datetime`/`date`/`UUID`/ 

133 `Path`/`Decimal`/`bytes` become strings, and `NaN` becomes `None`. A 

134 registered callable is portable only in a field the runner resolves names 

135 in — the `factory` fields and `store.filter` — since elsewhere its name 

136 reloads as a plain string. And a live registered object in scanner `params` 

137 is rejected even though scout would re-inflate it, since that only holds 

138 when every scanner entry is a spec reference. 

139 

140 A registered object's name must also be resolvable in the child. Inspect 

141 qualifies names of objects defined in an installed package 

142 (`my_pkg/my_task`), which the child can import; a bare name from a loose 

143 module relies on that module already being imported and is *not* checked 

144 here. Prefer factories from packaged code. 

145 

146 Args: 

147 spec: The flow spec to validate. 

148 

149 Raises: 

150 SpecNotPortableError: If the spec holds values that do not survive 

151 the boundary. The error's `violations` give the field path and 

152 problem for each one. 

153 """ 

154 violations: list[SpecViolation] = [] 

155 _walk_early_stopping(spec, "", violations) 

156 _walk(spec, "", violations) 

157 if violations: 

158 raise SpecNotPortableError(violations)

161def _walk_early_stopping( 

162 value: Any, 

163 path: str, 

164 violations: list[SpecViolation], 

165 seen: frozenset[int] = frozenset(), 

166 depth: int = 0, 

167) -> None: 

168 """Report every set `early_stopping`, wherever a FlowTask appears. 

169 

170 This needs its own pass because the rule is invisible to the dump: an 

171 `EarlyStopping` implementation that pydantic serializes natively (a 

172 dataclass or BaseModel) reloads as a plain dict, silently losing the 

173 protocol, so `_walk` would prune the subtree as clean. 

174 """ 

175 # This pass cannot prune, so it needs the same cycle guard `_walk` carries: 

176 # the depth limit bounds depth, not breadth, and a cycle whose container has 

177 # two or more branches re-enters itself from each one. 

178 if depth > _MAX_DEPTH or id(value) in seen: 

179 return 

180 if isinstance(value, FlowTask) and is_set(value.early_stopping): 

181 violations.append( 

182 SpecViolation(f"{path}.early_stopping".lstrip("."), _EARLY_STOPPING_MESSAGE) 

183 ) 

184 for segment, child in _children(value): 

185 _walk_early_stopping( 

186 child, path + segment, violations, seen | {id(value)}, depth + 1 

187 )

190def _offenders(value: Any, *, reinflated: bool, resolving: bool) -> list[Any]: 

191 """The values in this subtree that dumping coerces beyond recovery. 

192 

193 Returning the objects rather than a flag lets `_walk` tell "this node is 

194 itself the offender" from "the loss is somewhere below me", which decides 

195 whether to report here or keep descending. A subtree pydantic refuses to 

196 serialize at all yields the `_REFUSED` sentinel, but only when nothing else 

197 was recorded — a recorded value is the better explanation of the refusal. 

198 """ 

199 coerced: list[Any] = [] 

200 

201 def record(obj: Any) -> Any: 

202 coerced.append(obj) 

203 try: 

204 # Serialize as the real dump would, so that pydantic keeps walking 

205 # into a coerced value (e.g. a registry dict) and reports what is 

206 # nested inside it too. 

207 return serialize_fallback(obj) 

208 except Exception: 

209 # callable_name raises on callables with no __code__ (a partial or 

210 # callable object); the object is recorded, so it is still caught. 

211 return None 

212 

213 try: 

214 if isinstance(value, BaseModel): 

215 args = {**MODEL_DUMP_ARGS, "fallback": record} 

216 if isinstance(value, FlowTask): 

217 # Owned by _walk_early_stopping; including it here would report 

218 # the same stopper twice. 

219 args["exclude"] = {"early_stopping"} 

220 value.model_dump(**args) 

221 else: 

222 to_jsonable_python(value, fallback=record) 

223 refused = False 

224 except Exception: 

225 # Pydantic would not serialize this at all (a cycle, undecodable bytes, 

226 # excessive depth), so something here is non-portable even if nothing 

227 # reached the fallback first. 

228 refused = True 

229 offenders = [ 

230 obj 

231 for obj in coerced 

232 if not (resolving and survives_round_trip(obj)) 

233 and not (reinflated and serializes_to_registry_dict(obj)) 

234 ] 

235 if refused and not offenders: 

236 # Attribute to a marker rather than to `value`, so the walk still 

237 # descends to whichever leaf actually caused the refusal. Only when 

238 # nothing was recorded: dumping a subtree in isolation cannot apply 

239 # exclude_unset, so it can record a value the boundary drops, and 

240 # reporting the refusal here as well would blame this node for it. 

241 return [_REFUSED] 

242 return offenders

245def _children(value: Any) -> list[tuple[str, Any]]: 

246 """The addressable sub-values of a node, with the path segment for each.""" 

247 if isinstance(value, BaseModel): 

248 # model_fields_set mirrors the dump's exclude_unset: an unset field is 

249 # not serialized, so it cannot carry a value across the boundary. 

250 names = value.model_fields_set 

251 if isinstance(value, FlowTask): 

252 # _walk_early_stopping owns this field; descending would report a 

253 # live stopper a second time. 

254 names = names - {"early_stopping"} 

255 # Declaration order first, then any extra fields, so that sibling 

256 # violations render in a stable order rather than set-iteration order, 

257 # which varies with the hash seed. 

258 declared = [name for name in type(value).model_fields if name in names] 

259 extra = sorted(names - set(declared)) 

260 return [(f".{name}", getattr(value, name)) for name in declared + extra] 

261 if isinstance(value, Mapping): 

262 return [(f"[{key!r}]", item) for key, item in value.items()] 

263 if isinstance(value, Sequence) and not isinstance(value, (str, bytes)): 

264 return [(f"[{index}]", item) for index, item in enumerate(value)] 

265 return []

268def _message(value: Any) -> str: 

269 # The registry's own type is the discriminator: Scorer/Solver/Agent are 

270 # runtime-checkable Protocols that any callable satisfies structurally, so 

271 # isinstance would label a lambda a Scorer. 

272 if is_registry_object(value): 

273 return _REGISTRY_TYPE_MESSAGES.get(registry_info(value).type, _VALUE_MESSAGE) 

274 if isinstance(value, Task): 

275 return _TASK_MESSAGE 

276 if isinstance(value, Model): 

277 return _MODEL_MESSAGE 

278 if callable(value): 

279 return _CALLABLE_MESSAGE 

280 return _VALUE_MESSAGE

283def _walk( 

284 value: Any, 

285 path: str, 

286 violations: list[SpecViolation], 

287 *, 

288 reinflated: bool = False, 

289 resolving: bool = False, 

290 seen: frozenset[int] = frozenset(), 

291 depth: int = 0, 

292) -> None: 

293 """Report the non-portable values in a subtree, at the deepest path each. 

294 

295 The guards below run fatal-and-cheap first (depth), then the prune that makes 

296 the walk affordable, then the three cases where descending cannot help, then 

297 descent, then the post-descent fallback. 

298 """ 

299 if depth > _MAX_DEPTH: 

300 # Only reached below a node that already had offenders, so reporting here 

301 # without consulting _offenders again is warranted. 

302 violations.append(SpecViolation(path.lstrip("."), _VALUE_MESSAGE)) 

303 return 

304 children = _children(value) 

305 # `resolving` excuses a callable because the runner looks its name up, which 

306 # only applies to the callable itself -- not to a model sitting in the same 

307 # field, whose own `args` nobody resolves. Honour it at a leaf only. 

308 offenders = _offenders( 

309 value, reinflated=reinflated, resolving=resolving and not children 

310 ) 

311 if not offenders: 

312 return 

313 if id(value) in seen: 

314 # A cycle cannot be serialized at all; report here rather than recurse. 

315 violations.append(SpecViolation(path.lstrip("."), _VALUE_MESSAGE)) 

316 return 

317 if isinstance(value, Mapping) and any(not isinstance(key, str) for key in value): 

318 # Keys are not children, and a non-string key is coerced to text. 

319 violations.append(SpecViolation(path.lstrip("."), _VALUE_MESSAGE)) 

320 return 

321 if not children or any(offender is value for offender in offenders): 

322 # Either there is nowhere further to look, or this node is itself one of 

323 # the values that failed to serialize -- which a container can be even 

324 # when every child of it is portable (a `range`, a `memoryview`, a 

325 # custom Sequence or Mapping). 

326 violations.append(SpecViolation(path.lstrip("."), _message(value))) 

327 return 

328 before = len(violations) 

329 # Only a field of one of flow's own types earns an excuse below, since it is 

330 # flow's runner that re-inflates and looks up those fields. A foreign model 

331 # that happens to declare one of the same names -- `ScannerConfig.model_args` 

332 # and `ScannerConfig.filter` both do -- gets no excuse: nothing passes them 

333 # through `registry_kwargs` or a registry lookup. Keyed on the declaring 

334 # module rather than a list of types, so a type added to `_types` later is 

335 # covered without anyone remembering to update this. 

336 excused = type(value).__module__.startswith("inspect_flow._types") 

337 for segment, child in children: 

338 name = segment.lstrip(".") if excused else "" 

339 _walk( 

340 child, 

341 path + segment, 

342 violations, 

343 # Each child is judged in its own context, and the two flags spread 

344 # differently. `registry_kwargs` recurses to any depth, so anything 

345 # under a re-inflated field is re-inflated too -- the flag sticks. 

346 # A name lookup applies only to the value in the field itself, so 

347 # entering a new field replaces `resolving`; indexing into a 

348 # container keeps it. Without that, `FlowFactory.args` would inherit 

349 # the excuse from the `factory` field it hangs off. 

350 reinflated=reinflated or name in _REINFLATED_FIELDS, 

351 resolving=( 

352 name in _RESOLVING_FIELDS if segment.startswith(".") else resolving 

353 ), 

354 seen=seen | {id(value)}, 

355 depth=depth + 1, 

356 ) 

357 if len(violations) == before and any(o is _REFUSED for o in offenders): 

358 # Pydantic refused this subtree and no child accounts for it, so the node 

359 # itself is unserializable (a bytearray of undecodable bytes, whose 

360 # elements are individually fine). Compared by identity, as above: `in` 

361 # would fall back to `==` against arbitrary user values, and one whose 

362 # `__eq__` misbehaves (a numpy array's returns an array) would raise out 

363 # of the membership test. Deliberately limited to a refusal: 

364 # generalizing it rejects working specs -- see "Rejected alternatives" in 

365 # design/portable_spec_validation.md. 

366 violations.append(SpecViolation(path.lstrip("."), _message(value)))
